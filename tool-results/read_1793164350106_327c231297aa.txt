   158→function ViagemActiva({ detail, aoMudar, gpsOk }: { detail: Detail; aoMudar: () => void; gpsOk: string }) {
   159→  const { trip } = detail
   160→  const [busy, setBusy] = React.useState(false)
   161→  const [msg, setMsg] = React.useState<string | null>(null)
   162→  const [odo, setOdo] = React.useState('')
   163→  const [sos, setSos] = React.useState(false)
   164→  const [sosNota, setSosNota] = React.useState('')
   165→
   166→  const checklist = detail.checklist ? (JSON.parse(detail.checklist.itemsJson) as Array<{ key: string; label: string; blocking: boolean; ok?: boolean }>) : []
   167→  const [respostas, setRespostas] = React.useState<Record<string, boolean>>({})
   168→  React.useEffect(() => {
   169→    if (detail.checklist) {
   170→      const inicial: Record<string, boolean> = {}
   171→      for (const i of checklist) if (i.ok !== undefined) inicial[i.key] = i.ok
   172→      setRespostas(inicial)
   173→    }
   174→  }, [detail.checklist?.itemsJson])  
   175→
   176→  const agir = async (action: string, extra: Record<string, unknown> = {}) => {
   177→    setBusy(true); setMsg(null)
   178→    try {
   179→      await api(`/api/trips/${trip.id}/actions`, { method: 'POST', body: { action, ...extra } })
   180→      aoMudar()
   181→      setMsg('Registado no sistema.')
   182→    } catch (e) { setMsg(e instanceof Error ? e.message : 'Falha na acção.') }
   183→    finally { setBusy(false) }
   184→  }
   185→
   186→  const submeterChecklist = async () => {
   187→    if (!detail.checklist) return
   188→    setBusy(true); setMsg(null)
   189→    try {
   190→      const r = await api<{ passed: boolean; message: string }>(`/api/trips/${trip.id}/checklist`, {
   191→        method: 'POST', body: { items: checklist.map((i) => ({ key: i.key, ok: respostas[i.key] })) },
   192→      })
   193→      setMsg(r.message); aoMudar()
   194→    } catch (e) { setMsg(e instanceof Error ? e.message : 'Falha ao submeter checklist.') }
   195→    finally { setBusy(false) }
   196→  }
   197→
   198→  const enviarSos = async () => {
   199→    setBusy(true)
   200→    try {
   201→      const pos = gpsOk === 'ok' ? await obterPosicao() : null
   202→      const uuid = novoUuid()
   203→      // SOS funciona OFFLINE: entra na fila com hora original do dispositivo (OFF-001)
   204→      await filaGuardar({
   205→        clientUuid: uuid, type: 'TRIP_EVENT', deviceTime: new Date().toISOString(),
   206→        payload: { tripId: trip.id, type: 'SOS', note: sosNota || 'Pedido de ajuda (SOS)', lat: pos?.lat, lng: pos?.lng, locSource: pos ? 'BROWSER' : 'NONE', locStatus: pos ? 'OK' : 'UNAVAILABLE' },
   207→        estadoLocal: 'GUARDADO', tentativas: 0, resumo: `SOS — ${trip.code}`,
   208→      })
   209→      setSos(false); setSosNota('')
   210→      setMsg('SOS registado. Será enviado assim que houver internet.')
   211→    } finally { setBusy(false) }
   212→  }
   213→
   214→  const STATUS: Record<string, string> = {
   215→    AWAITING_CHECKLIST: 'A aguardar checklist', AUTHORIZED_DEPARTURE: 'Autorizada para partida',
   216→    IN_TRANSIT: 'Em trânsito', PAUSED: 'Em pausa / retida', ARRIVED: 'Chegada registada',
   217→  }
   218→
   219→  return (
   220→    <>
   221→      <Card>
   222→        <CardContent className="pt-5 space-y-3">
   223→          <div className="flex items-start justify-between gap-2">
   224→            <div>
   225→              <div className="text-xs text-muted-foreground">Viagem activa</div>
   226→              <div className="text-lg font-bold">{trip.code}</div>
   227→            </div>
   228→            <Badge className="bg-teal-700 text-white">{STATUS[trip.status] ?? trip.status}</Badge>
   229→          </div>
   230→          <div className="text-sm">{trip.origin} <ChevronRight className="inline h-4 w-4 text-zinc-400" /> {trip.destination}</div>
   231→          {trip.cargoDesc && <div className="text-xs text-muted-foreground">Carga: {trip.cargoDesc}{trip.client ? ` • Cliente: ${trip.client}` : ''}</div>}
   232→          {msg && <div className="rounded-md border border-teal-300 bg-teal-50 px-3 py-2 text-sm text-teal-900">{msg}</div>}
   233→        </CardContent>
   234→      </Card>
   235→
   236→      {/* Accões rápidas (§11.2: despesa, foto, incidente, checklist, SOS) */}
   237→      {trip.status === 'AWAITING_CHECKLIST' && (
   238→        <Card>
   239→          <CardContent className="pt-5 space-y-2.5">
   240→            <div className="font-semibold text-sm">Checklist pré-partida — confirme cada item</div>
   241→            <p className="text-xs text-muted-foreground">O carregador de telemóvel é obrigatório: sem ele a viagem não é autorizada (requisito crítico).</p>
   242→            {checklist.map((i) => (
   243→              <label key={i.key} className={`flex items-start gap-2.5 rounded-lg border p-3 text-sm ${respostas[i.key] === false && i.blocking ? 'border-red-300 bg-red-50' : ''}`}>
   244→                <Checkbox className="mt-0.5" checked={respostas[i.key] ?? false} onCheckedChange={(v) => setRespostas({ ...respostas, [i.key]: v === true })} />
   245→                <span>{i.label}{i.blocking && <b className="ml-1 text-red-700 text-xs">(obrigatório)</b>}</span>
   246→              </label>
   247→            ))}
   248→            <Button className="w-full min-h-[48px] bg-teal-700 hover:bg-teal-600" disabled={busy || checklist.some((i) => respostas[i.key] === undefined)} onClick={submeterChecklist}>
   249→              Submeter checklist
   250→            </Button>
   251→          </CardContent>
   252→        </Card>
   253→      )}
   254→
   255→      {trip.status === 'AUTHORIZED_DEPARTURE' && (
   256→        <Card>
   257→          <CardContent className="pt-5 space-y-3">
   258→            <div className="font-semibold text-sm">Registar partida</div>
   259→            <div className="space-y-1.5">
   260→              <Label htmlFor="odo">Odómetro de partida (km)</Label>
   261→              <Input id="odo" type="number" inputMode="numeric" value={odo} onChange={(e) => setOdo(e.target.value)} className="min-h-[48px] text-lg" />
   262→            </div>
   263→            <Button className="w-full min-h-[48px] bg-teal-700 hover:bg-teal-600" disabled={busy || !odo} onClick={() => agir('depart', { departureOdometerKm: Number(odo) })}>
   264→              Confirmar partida
   265→            </Button>
   266→          </CardContent>
   267→        </Card>
   268→      )}
   269→
   270→      {['IN_TRANSIT', 'PAUSED'].includes(trip.status) && (
   271→        <Card>
   272→          <CardContent className="pt-5 grid grid-cols-2 gap-2">
   273→            {trip.status === 'IN_TRANSIT' ? (
   274→              <Button variant="outline" className="min-h-[48px]" disabled={busy} onClick={() => { const motivo = window.prompt('Motivo da paragem:'); if (motivo) agir('pause', { reason: motivo }) }}>Registar pausa</Button>
   275→            ) : (
   276→              <Button className="min-h-[48px] bg-teal-700 hover:bg-teal-600" disabled={busy} onClick={() => agir('resume')}>Retomar viagem</Button>
   277→            )}
   278→            <Button variant="outline" className="min-h-[48px]" disabled={busy} onClick={() => { const odoC = window.prompt('Odómetro de chegada (km):'); if (odoC) agir('arrive', { arrivalOdometerKm: Number(odoC) }) }}>Chegada ao destino</Button>
   279→          </CardContent>
   280→        </Card>
   281→      )}
   282→
   283→      {trip.status === 'ARRIVED' && (
   284→        <Card><CardContent className="pt-5 text-sm text-muted-foreground">Chegada registada. Aguarde a revisão do escritório — o fecho é feito pela equipa após conferência de despesas (FR-038).</CardContent></Card>
   285→      )}
   286→
   287→      <Button variant="outline" className="w-full min-h-[48px] border-red-300 text-red-700 hover:bg-red-50" onClick={() => setSos(true)}>
   288→        <Siren className="mr-2 h-4 w-4" />SOS — pedir ajuda
   289→      </Button>
   290→      <Confirmacao open={sos} onOpenChange={setSos} perigo exigirTexto="SOS"
   291→        titulo="Confirmar SOS"
   292→        mensagem="Serão enviados a viagem, coordenadas disponíveis e a hora do dispositivo. Confirme apenas em situação real — uso acidental mobiliza resposta (GPS-026)."
   293→        onConfirm={enviarSos} />
   294→    </>
   295→  )
   296→}

   298→function DespesasMotorista({ detail, aoMudar, refrescarFila, gpsOk, online }: {
   299→  detail: Detail | null; aoMudar: () => void; refrescarFila: () => void; gpsOk: string; online: boolean
   300→}) {
   301→  const [aberto, setAberto] = React.useState(false)
   302→  const [form, setForm] = React.useState({ categoria: 'combustivel', valor: '', moeda: 'MZN', litros: '', preco: '', posto: '', descricao: '', semComprovativo: '' })
   303→  const [foto, setFoto] = React.useState<{ dataUrl: string; width: number; height: number } | null>(null)
   304→  const [busy, setBusy] = React.useState(false)
   305→  const [msg, setMsg] = React.useState<string | null>(null)
   306→  const [erro, setErro] = React.useState<string | null>(null)
   307→
   308→  const guardar = async (enviar: boolean) => {
   309→    if (!detail) return
   310→    setBusy(true); setErro(null); setMsg(null)
   311→    try {
   312→      const valorMinor = parseMT(form.valor)
   313→      if (valorMinor <= 0) throw new Error('Indique o valor da despesa.')
   314→      const pos = gpsOk === 'ok' ? await obterPosicao() : null
   315→      const uuid = novoUuid()
   316→      const agora = new Date().toISOString()
   317→      await filaGuardar({
   318→        clientUuid: uuid, type: 'EXPENSE', deviceTime: agora,
   319→        payload: {
   320→          tripId: detail.trip.id, category: form.categoria, amountMinor: valorMinor, currency: form.moeda,
   321→          expenseDate: agora, vendor: form.posto || undefined, description: form.descricao || undefined,
   322→          liters: form.litros ? Number(form.litros) : undefined,
   323→          pricePerLiterMinor: form.preco ? Math.round(Number(form.preco.replace(',', '.')) * 100) : undefined,
   324→          station: form.posto || undefined,
   325→          lat: pos?.lat, lng: pos?.lng, locPrecisionM: pos?.accuracyM,
   326→          locSource: pos ? 'BROWSER' : 'NONE',
   327→          locStatus: pos ? 'OK' : 'UNAVAILABLE',
   328→          locNote: pos ? undefined : (gpsOk === 'negado' ? 'Permissão de localização negada' : 'Localização indisponível no momento do registo'),
   329→          noReceiptReason: form.semComprovativo || undefined,
   330→          dataUrl: foto?.dataUrl, width: foto?.width, height: foto?.height, capturedAt: agora,
   331→        },
   332→        estadoLocal: 'GUARDADO', tentativas: 0,
   333→        resumo: `${CATS.find((c) => c.value === form.categoria)?.label}: ${fmtMT(valorMinor, form.moeda)}`,
   334→      })
   335→      // tentar sincronizar imediatamente quando há rede (OFF-005)
   336→      if (online && enviar) {
   337→        const { sincronizar } = await import('@/lib/client/offline')
   338→        await sincronizar()
   339→      }
   340→      setForm({ categoria: 'combustivel', valor: '', moeda: 'MZN', litros: '', preco: '', posto: '', descricao: '', semComprovativo: '' })
   341→      setFoto(null); setAberto(false)
   342→      setMsg(online ? 'Despesa guardada e submetida.' : 'Despesa guardada NESTE TELEMÓVEL — será enviada automaticamente quando houver internet.')
   343→      refrescarFila(); aoMudar()
   344→    } catch (e) { setErro(e instanceof Error ? e.message : 'Falha ao guardar despesa.') }
   345→    finally { setBusy(false) }
   346→  }
   347→
   348→  const pegarFoto = async (e: React.ChangeEvent<HTMLInputElement>) => {
   349→    const file = e.target.files?.[0]
   350→    if (!file) return
   351→    try {
   352→      const compressed = await comprimirImagem(file)
   353→      setFoto(compressed)
   354→    } catch { setErro('Não foi possível comprimir a fotografia.') }
   355→  }
   356→
   357→  const detalhes = detail?.expenses ?? []
   358→
   359→  return (
   360→    <>
   361→      <div className="grid grid-cols-2 gap-2">
   362→        <Button className="min-h-[48px] bg-teal-700 hover:bg-teal-600" onClick={() => setAberto(true)}><Camera className="mr-2 h-4 w-4" />Nova despesa + foto</Button>
   363→        <Button variant="outline" className="min-h-[48px]" onClick={() => setAberto(true)}>Despesa sem câmara</Button>
   364→      </div>
   365→
   366→      <Card>

   460→function FilaSincronizacao({ fila, refrescar }: { fila: FilaItem[]; refrescar: () => void }) {
   461→  const [busy, setBusy] = React.useState(false)
