   188→function Dispositivos({ podeGerir }: { podeGerir: boolean }) {
   189→  const { sessao } = useApp()
   190→  const [rows, setRows] = React.useState<Device[]>([])
   191→  const [vehicles, setVehicles] = React.useState<Vehicle[]>([])
   192→  const [erro, setErro] = React.useState<string | null>(null)
   193→  const [novo, setNovo] = React.useState(false)
   194→  const [form, setForm] = React.useState<Record<string, string>>({})
   195→  const [novaChave, setNovaChave] = React.useState<string | null>(null)
   196→  const [confirmarRotacao, setConfirmarRotacao] = React.useState<Device | null>(null)
   197→  const isDemo = sessao.company?.isDemo
   198→
   199→  const carregar = React.useCallback(async () => {
   200→    try {
   201→      const [d, v] = await Promise.all([api<{ devices: Device[] }>('/api/gps/devices'), api<{ vehicles: Vehicle[] }>('/api/vehicles')])
   202→      setRows(d.devices); setVehicles(v.vehicles); setErro(null)
   203→    } catch (e) { setErro(e instanceof Error ? e.message : 'Falha ao carregar dispositivos.') }
   204→  }, [])
   205→  React.useEffect(() => { carregar() }, [carregar])
   206→
   207→  const criar = async () => {
   208→    try { await api('/api/gps/devices', { method: 'POST', body: form }); setNovo(false); setForm({}); carregar() }
   209→    catch (e) { setErro(e instanceof Error ? e.message : 'Falha ao criar dispositivo.') }
   210→  }
   211→
   212→  const atribuir = async (device: Device, vehicleId: string) => {
   213→    try { await api('/api/gps/devices', { method: 'PATCH', body: { id: device.id, action: 'assign', vehicleId } }); carregar() }
   214→    catch (e) { setErro(e instanceof Error ? e.message : 'Falha ao associar.') }
   215→  }
   216→
   217→  const rodarChave = async (device: Device) => {
   218→    try {
   219→      const r = await api<{ apiKey: string }>('/api/gps/devices', { method: 'PATCH', body: { id: device.id, action: 'rotate-key' } })
   220→      setNovaChave(r.apiKey); setConfirmarRotacao(null)
   221→    } catch (e) { setErro(e instanceof Error ? e.message : 'Falha ao rodar chave.') }
   222→  }
   223→
   224→  const copiarIngest = async () => {
   225→    await navigator.clipboard.writeText(JSON.stringify({ points: [{ deviceSerial: rows[0]?.serial ?? 'TLT-DEMO-001', lat: -19.8436, lng: 34.8389, speedKmh: 62, heading: 270, accuracyM: 8, gnssTime: new Date().toISOString(), seq: 10231, ignition: true, fixQuality: '3D' }] }, null, 2))
   226→  }
   227→
   228→  return (
   229→    <div className="space-y-4">
   230→      <ErroAlerta erro={erro} />
   231→      <Panel title="Integração de telemetria (GPS-008)" description="Camada de adaptadores: o fornecedor real é convertido para o contrato comum sem alterar os módulos de viagem/despesas."
   232→        actions={isDemo && <Button size="sm" variant="outline" onClick={copiarIngest}><FileUp className="mr-1 h-4 w-4" />Copiar payload de teste</Button>}>
   233→        <div className="grid gap-3 text-sm sm:grid-cols-3">
   234→          <div className="rounded-lg border p-3"><b className="block text-xs uppercase text-muted-foreground">Endpoint de ingestão</b><code className="text-xs">POST /api/gps/ingest</code><p className="mt-1 text-xs text-muted-foreground">Autenticação: Bearer apiKey do dispositivo. Lotes store-and-forward com seq deduplicada.</p></div>
   235→          <div className="rounded-lg border p-3"><b className="block text-xs uppercase text-muted-foreground">Adaptadores</b><code className="text-xs">generic-json</code><p className="mt-1 text-xs text-muted-foreground">Fornecedor real (Teltonika, Queclink…) acrescenta o seu adapter — homologação pendente (ARC-005).</p></div>
   236→          <div className="rounded-lg border p-3"><b className="block text-xs uppercase text-muted-foreground">Documentação viva</b><code className="text-xs">GET /api/gps/ingest</code><p className="mt-1 text-xs text-muted-foreground">Contrato, exemplo de payload e regras de deduplicação/histórico.</p></div>
   237→        </div>
   238→      </Panel>
   239→
   240→      <Panel title={`Rastreadores (${rows.length})`} description="Fonte primária de posição — um por camião (GPS-001); chave por dispositivo com rotação (§14.3)"
   241→        actions={podeGerir && <Button size="sm" onClick={() => setNovo(true)}><Plus className="mr-1 h-4 w-4" />Novo dispositivo</Button>}>
   242→        <DataTable rows={rows} columns={[
   243→          { key: 'serial', header: 'Série/IMEI', render: (d) => <div><span className="font-medium">{d.serial}</span><div className="text-xs text-muted-foreground">{d.imei}</div></div> },
   244→          { key: 'vendor', header: 'Fornecedor', render: (d) => d.vendor ?? '—', hideOnMobile: true },
   245→          { key: 'vehicle', header: 'Veículo', render: (d) => d.vehicle ? <span className="font-medium">{d.vehicle.plate}</span> : <span className="text-muted-foreground">Sem associação</span> },
   246→          { key: 'sim', header: 'SIM', render: (d) => d.simNumber ?? '—', hideOnMobile: true },
   247→          { key: 'last', header: 'Último ponto', render: (d) => fmtData(d.lastPointAt, true), hideOnMobile: true },
   248→          { key: 'status', header: 'Estado', render: (d) => <StatusBadge estado={d.status} mapa={{}} /> },
   249→          { key: 'actions', header: '', render: (d) => podeGerir && (
   250→            <div className="flex gap-1.5">
   251→              <Selecao value={d.vehicleId ?? ''} onChange={(v) => v && atribuir(d, v)} opcoes={[{ value: '', label: 'Associar a…' }, ...vehicles.map((v) => ({ value: v.id, label: v.plate }))]} />
   252→              <Button size="sm" variant="outline" title="Rodar chave de API" onClick={() => setConfirmarRotacao(d)}><KeyRound className="h-3.5 w-3.5" /></Button>
   253→            </div>
   254→          ) },
   255→        ]} />
   256→      </Panel>
   257→
   258→      <Modal open={novo} onOpenChange={setNovo} title="Novo rastreador GPS" description="Registe a instalação (local, instalador) para validação por técnico (GPS-006)."
   259→        footer={<><Button variant="outline" onClick={() => setNovo(false)}>Cancelar</Button><Button onClick={criar}>Registar</Button></>}>
   260→        <div className="grid gap-3 sm:grid-cols-2">

   282→function Documentos({ podeGerir }: { podeGerir: boolean }) {
   283→  const [rows, setRows] = React.useState<Doc[]>([])
   284→  const [vehicles, setVehicles] = React.useState<Vehicle[]>([])
   285→  const [drivers, setDrivers] = React.useState<Driver[]>([])
   286→  const [erro, setErro] = React.useState<string | null>(null)
   287→  const [novo, setNovo] = React.useState(false)
   288→  const [form, setForm] = React.useState<Record<string, string>>({})
   289→  const [filtro, setFiltro] = React.useState('')
   290→
   291→  const carregar = React.useCallback(async () => {
   292→    try {
   293→      const [d, v, m] = await Promise.all([api<{ documents: Doc[] }>('/api/documents'), api<{ vehicles: Vehicle[] }>('/api/vehicles'), api<{ drivers: Driver[] }>('/api/drivers')])
   294→      setRows(d.documents); setVehicles(v.vehicles); setDrivers(m.drivers); setErro(null)
   295→    } catch (e) { setErro(e instanceof Error ? e.message : 'Falha ao carregar documentos.') }
   296→  }, [])
   297→  React.useEffect(() => { carregar() }, [carregar])
   298→
   299→  const criar = async () => {
   300→    try {
   301→      await api('/api/documents', { method: 'POST', body: { ...form, ownerType: form.ownerType || 'VEHICLE', vehicleId: form.ownerType === 'VEHICLE' ? form.ownerId : undefined, driverId: form.ownerType === 'DRIVER' ? form.ownerId : undefined, blocking: form.blocking === '1' } })
   302→      setNovo(false); setForm({}); carregar()
   303→    } catch (e) { setErro(e instanceof Error ? e.message : 'Falha ao criar documento.') }
   304→  }
   305→
   306→  const TIPOS = [
   307→    { value: 'seguro', label: 'Seguro' }, { value: 'inspecao', label: 'Inspecção' },
   308→    { value: 'licenca', label: 'Licença' }, { value: 'autorizacao_transfronteira', label: 'Autorização transfronteiriça' },
   309→    { value: 'carta_conducao', label: 'Carta de condução' }, { value: 'aduaneiro', label: 'Documento aduaneiro' },
   310→    { value: 'certificado_carga', label: 'Certificado de carga' }, { value: 'outro', label: 'Outro' },
   311→  ]
   312→
   313→  return (
   314→    <div className="space-y-4">
   315→      <ErroAlerta erro={erro} />
   316→      <div className="flex flex-wrap gap-2">
   317→        {[{ k: '', l: 'Todos' }, { k: 'expiring', l: 'A expirar (30 dias)' }, { k: 'expired', l: 'Expirados' }].map((f) => (
   318→          <Button key={f.k} size="sm" variant={filtro === f.k ? 'default' : 'outline'} onClick={() => { setFiltro(f.k); api<{ documents: Doc[] }>(`/api/documents${f.k ? `?filter=${f.k}` : ''}`).then((r) => setRows(r.documents)) }}>{f.l}</Button>
   319→        ))}
   320→        {podeGerir && <Button size="sm" className="ml-auto" onClick={() => setNovo(true)}><Plus className="mr-1 h-4 w-4" />Novo documento</Button>}
   321→      </div>
   322→      <Panel title={`Documentos (${rows.length})`} description="Vencimentos com alerta; bloqueantes expirados impedem a partida (DOC-002, AT-017)">
   323→        <DataTable rows={rows} columns={[
   324→          { key: 'type', header: 'Tipo', render: (d) => <span className="font-medium">{TIPOS.find((t) => t.value === d.type)?.label ?? d.type}</span> },
   325→          { key: 'number', header: 'Número', hideOnMobile: true },
   326→          { key: 'owner', header: 'Associado a', render: (d) => d.ownerType === 'VEHICLE' ? (vehicles.find((v) => v.id === d.vehicleId)?.plate ?? '—') : d.ownerType === 'DRIVER' ? (drivers.find((x) => x.id === d.driverId)?.name ?? '—') : d.ownerType },
   327→          { key: 'expiresAt', header: 'Validade', render: (d) => <span className={d.status === 'EXPIRED' ? 'text-red-600 font-medium' : d.status === 'EXPIRING' ? 'text-amber-600' : ''}>{fmtData(d.expiresAt)}</span> },
   328→          { key: 'blocking', header: 'Bloqueante', render: (d) => d.blocking ? <Badge variant="outline" className="border-red-300 text-red-700">Sim — bloqueia partida</Badge> : 'Não' },
   329→          { key: 'status', header: 'Estado', render: (d) => <StatusBadge estado={d.status} mapa={{ VALID: 'Válido', EXPIRING: 'A expirar', EXPIRED: 'Expirado' }} /> },
   330→          { key: 'country', header: 'País', render: (d) => d.issuerCountry ?? '—', hideOnMobile: true },
   331→        ]} />
   332→      </Panel>
   333→
   334→      <Modal open={novo} onOpenChange={setNovo} title="Novo documento" description="Lista configurável por rota/país — a completude legal exige validação de especialista local (DOC-003)."
   335→        footer={<><Button variant="outline" onClick={() => setNovo(false)}>Cancelar</Button><Button onClick={criar}>Registar</Button></>}>
   336→        <div className="grid gap-3 sm:grid-cols-2">
   337→          <Campo label="Associado a" required><Selecao value={form.ownerType ?? 'VEHICLE'} onChange={(v) => setForm({ ...form, ownerType: v })} opcoes={[{ value: 'VEHICLE', label: 'Veículo' }, { value: 'DRIVER', label: 'Motorista' }]} /></Campo>
   338→          <Campo label={form.ownerType === 'DRIVER' ? 'Motorista' : 'Veículo'} required>
   339→            <Selecao value={form.ownerId ?? ''} onChange={(v) => setForm({ ...form, ownerId: v })} opcoes={[{ value: '', label: 'Seleccionar…' }, ...(form.ownerType === 'DRIVER' ? drivers.map((d) => ({ value: d.id, label: d.name })) : vehicles.map((v) => ({ value: v.id, label: v.plate })))]} />
   340→          </Campo>
   341→          <Campo label="Tipo" required><Selecao value={form.type ?? 'seguro'} onChange={(v) => setForm({ ...form, type: v })} opcoes={TIPOS} /></Campo>

   347→        </div>

   349→    </div>
